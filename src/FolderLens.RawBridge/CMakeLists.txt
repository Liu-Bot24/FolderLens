cmake_minimum_required(VERSION 3.24)
project(FolderLensRawBridge LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "FolderLens RawBridge requires Windows x64")
endif()
set(LIBRAW_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.tools/libraw/LibRaw-0.22.2" CACHE PATH "Verified LibRaw distribution")
add_library(FolderLens.RawBridge SHARED RawBridge.cpp)
target_compile_features(FolderLens.RawBridge PRIVATE cxx_std_17)
target_compile_options(FolderLens.RawBridge PRIVATE /EHsc /W4)
target_compile_definitions(FolderLens.RawBridge PRIVATE LIBRAW_BUILDLIB _CRT_SECURE_NO_WARNINGS)
target_include_directories(FolderLens.RawBridge PRIVATE "${LIBRAW_ROOT}")
target_link_libraries(FolderLens.RawBridge PRIVATE "${LIBRAW_ROOT}/lib/libraw.lib")
add_custom_command(TARGET FolderLens.RawBridge POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${LIBRAW_ROOT}/bin/libraw.dll" "$<TARGET_FILE_DIR:FolderLens.RawBridge>")
